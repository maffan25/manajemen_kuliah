/**
 * BEM STDIIS | Samasta Corner: penerima absensi dan penilaian event
 * Tempel seluruh kode ini di Google Sheets: Ekstensi > Apps Script.
 * Gunakan Spreadsheet SENDIRI untuk Corner (bukan Sheet Aspirasi), karena data ini
 * memuat nama dan NIM peserta. Panduan lengkap: PANDUAN-CORNER.txt
 */
var SHEET_ID = '';            // OPSIONAL. Isi ID Spreadsheet bila skrip dibuat terpisah dari Sheet.
var SHEET_NAME = 'Absensi';
var KODE_AKSES = '';          // WAJIB DIISI: kode rahasia Irjen untuk membuka Pembukuan. Minimal 6 karakter.
var MAX_LEN = 1000;
var MAX_JAWAB = 30;           // jumlah maksimal jawaban per pengisi

/* ===== Peserta mengirim absensi dan penilaian ===== */
function doPost(e) {
  try {
    var raw = (e && e.postData && e.postData.contents) || '';
    var d = JSON.parse(raw);

    // Jebakan bot: kolom tersembunyi ini hanya terisi oleh bot
    if (d.website) return reply_({ ok: true });
    if (d.aksi === 'hasil') return hasil_(d);           // Irjen membaca pembukuan (kode dikirim lewat POST, tidak muncul di URL)
    if (d.aksi !== 'absen') return reply_({ ok: false, error: 'aksi' });

    var corner = id_(d.corner), event = id_(d.event);
    var nama = clean_(d.nama, 80), nim = clean_(d.nim, 20).toUpperCase(), bagian = clean_(d.bagian, 80);
    if (!corner || !event || nama.length < 3 || !/^[0-9A-Z.\-]{5,20}$/.test(nim) || !bagian) return reply_({ ok: false, error: 'data' });

    // Jawaban: objek {idSoal: angka 1..5 ATAU teks}. Soal bintang harus bilangan bulat 1..5.
    var jawab = {}, n = 0, k;
    for (k in (d.jawab || {})) {
      var id = id_(k), raw = d.jawab[k];
      if (!id) return reply_({ ok: false, error: 'skor' });
      if (typeof raw === 'number') {
        if (!(raw >= 1 && raw <= 5) || Math.floor(raw) !== raw) return reply_({ ok: false, error: 'skor' });
        jawab[id] = raw;
      } else {
        var t = clean_(raw, MAX_LEN); if (t) jawab[id] = t;
      }
      if (++n > MAX_JAWAB) return reply_({ ok: false, error: 'skor' });
    }
    // Ringkasan bintang per teks soal (agar kolom Skor di Sheet mudah dibaca)
    var skor = {};
    for (k in (d.skor || {})) { var v = Number(d.skor[k]); if (v >= 1 && v <= 5 && Math.floor(v) === v) skor[clean_(k, 60)] = v; }
    if (!Object.keys(jawab).length && !Object.keys(skor).length) return reply_({ ok: false, error: 'skor' });

    var sh = sheet_();
    if (!sh) return reply_({ ok: false, error: 'nosheet' });

    var lock = LockService.getScriptLock();
    lock.waitLock(20000);
    try {
      // Satu NIM hanya boleh mengisi satu kali per event
      var last = sh.getLastRow();
      if (last > 1) {
        var ids = sh.getRange(2, 3, last - 1, 3).getValues(); // kolom C (event), D (judul), E (NIM)
        for (var i = 0; i < ids.length; i++) {
          if (String(ids[i][0]) === event && String(ids[i][2]).toUpperCase() === nim) return reply_({ ok: false, error: 'duplikat' });
        }
      }
      sh.appendRow([new Date(), corner, event, clean_(d.judul, 160), nim, nama, bagian, JSON.stringify(skor), clean_(d.komentar, MAX_LEN), clean_(d.kritik, MAX_LEN), JSON.stringify(jawab)]);
      SpreadsheetApp.flush();
    } finally {
      lock.releaseLock();
    }
    return reply_({ ok: true });
  } catch (err) {
    console.error(err);
    return reply_({ ok: false, error: 'server' });
  }
}

/* ===== Irjen membaca pembukuan (wajib kode akses, dikirim lewat POST) ===== */
function hasil_(d) {
  var cache = CacheService.getScriptCache(), fails = Number(cache.get('gagal') || 0);
  if (fails >= 10) return reply_({ ok: false, error: 'tunggu' });            // ditahan 10 menit setelah 10 kali salah
  if (!KODE_AKSES || KODE_AKSES.length < 6) return reply_({ ok: false, error: 'belumkode' });
  if (String(d.kode || '') !== KODE_AKSES) {
    cache.put('gagal', String(fails + 1), 600);
    return reply_({ ok: false, error: 'kode' });
  }
  var event = id_(d.event), sh = sheet_();
  if (!event) return reply_({ ok: false, error: 'data' });
  if (!sh) return reply_({ ok: false, error: 'nosheet' });
  var last = sh.getLastRow(), rows = [];
  if (last > 1) {
    var vals = sh.getRange(2, 1, last - 1, 11).getValues();
    vals.forEach(function (r) {
      if (String(r[2]) !== event) return;
      var skor = {}, jawab = null;
      try { skor = JSON.parse(r[7]); } catch (x) {}
      try { if (r[10]) jawab = JSON.parse(r[10]); } catch (x) {}
      var row = {
        t: r[0] instanceof Date ? r[0].toISOString() : String(r[0]),
        nim: String(r[4]), nama: String(r[5]), bagian: String(r[6]),
        skor: skor, komentar: String(r[8] || ''), kritik: String(r[9] || '')
      };
      if (jawab) row.jawab = jawab;      // baris lama (sebelum soal fleksibel) tidak punya jawab
      rows.push(row);
    });
  }
  return reply_({ ok: true, rows: rows });
}

/* ===== Buka URL Web App di browser: harus muncul "ok":true DAN "sheet":"terhubung" ===== */
function doGet() {
  try {
    var sh2 = null, msg = '';
    try { sh2 = sheet_(); } catch (err) { msg = String(err); }
    return reply_({
      ok: true,
      info: 'Endpoint Samasta Corner aktif',
      sheet: sh2 ? 'terhubung' : 'TIDAK terhubung ke Spreadsheet',
      kode: (KODE_AKSES && KODE_AKSES.length >= 6) ? 'sudah diatur' : 'BELUM diatur (isi KODE_AKSES)',
      detail: msg
    });
  } catch (err2) {
    console.error(err2);
    return reply_({ ok: false, error: 'server' });
  }
}

// Jalankan SEKALI secara manual (pilih "setup" lalu klik Jalankan) untuk memberi izin
// dan membuat tab "Absensi". Setelah itu lakukan Terapkan > Deployment baru.
function setup() {
  var sh = sheet_();
  if (!sh) throw new Error('Tidak ada Spreadsheet. Pasang kode lewat Ekstensi > Apps Script dari dalam Sheet, atau isi SHEET_ID.');
  Logger.log('OK, tab "' + sh.getName() + '" siap di Spreadsheet: ' + sh.getParent().getUrl());
}

function clean_(v, max) {
  return String(v == null ? '' : v)
    .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '')
    .trim()
    .slice(0, max);
}

function id_(v) {
  var s = clean_(v, 80);
  return /^[A-Za-z0-9_\-]+$/.test(s) ? s : '';
}

function sheet_() {
  var ss = SHEET_ID ? SpreadsheetApp.openById(SHEET_ID) : SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) return null;
  var sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) {
    sh = ss.insertSheet(SHEET_NAME);
    sh.appendRow(['Waktu', 'Corner', 'EventID', 'Event', 'NIM', 'Nama', 'Bagian', 'Skor', 'Komentar', 'Kritik', 'Jawaban']);
    sh.setFrozenRows(1);
    sh.getRange('1:1').setFontWeight('bold');
    // Semua kolom teks diformat sebagai teks agar isi tidak pernah dibaca sebagai rumus
    sh.getRange('B:K').setNumberFormat('@');
    sh.getRange('A:A').setNumberFormat('dd/MM/yyyy HH:mm:ss');
    sh.getRange('I:J').setWrap(true);
    sh.setColumnWidth(1, 150);
    sh.setColumnWidth(4, 240);
    sh.setColumnWidth(6, 200);
    sh.setColumnWidth(8, 260);
    sh.setColumnWidth(9, 380);
    sh.setColumnWidth(10, 380);
    sh.setColumnWidth(11, 420);
  } else if (!sh.getRange(1, 11).getValue()) {
    // Sheet versi lama: tambahkan kolom Jawaban (K) tanpa mengubah data yang sudah ada
    sh.getRange(1, 11).setValue('Jawaban').setFontWeight('bold');
    sh.getRange('K:K').setNumberFormat('@');
    sh.setColumnWidth(11, 420);
  }
  return sh;
}

function reply_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
